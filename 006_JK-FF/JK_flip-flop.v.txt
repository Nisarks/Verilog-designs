module JK_FF(Q,J,K,CLK,RST);
output reg Q;
input J,K;
input CLK,RST;

always@(posedge CLK,posedge RST) begin
if(RST)
  Q<=1'b0;
else
 case({J,K})
  2'b00:Q<=Q;
  2'b01:Q<=1'b0;
  2'b10:Q<=1'b1;
  2'b11:Q<=~Q;
  endcase
end
endmodule
